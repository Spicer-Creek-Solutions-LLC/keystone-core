================================================================================
                               GETTING STARTED
================================================================================

1. Get the application URL:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
   HTTP:  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}
{{- end }}
{{- end }}
{{- if .Values.grpcIngress.enabled }}
{{- range $host := .Values.grpcIngress.hosts }}
   gRPC:  {{ $host.host }}:443
{{- end }}
{{- end }}
{{- if not (or .Values.ingress.enabled .Values.grpcIngress.enabled) }}

   # Port-forward to access locally:
   kubectl port-forward svc/{{ include "kscore-server.fullname" . }} 8080:{{ .Values.service.http.port }} 9090:{{ .Values.service.grpc.port }}

   # Access points:
   HTTP API:  http://localhost:8080
   gRPC API:  localhost:9090
{{- end }}

2. Check health status:
   curl http://localhost:8080/health/ready

3. Connect agents:
   # Install the agent with:
   helm install kscore-agent ./kscore-agent \
     --set server.address={{ include "kscore-server.fullname" . }}.{{ .Release.Namespace }}.svc.cluster.local:{{ .Values.service.grpc.port }}
